cmake_minimum_required(VERSION 3.22)
project(WebEngineQuestConsumer LANGUAGES CXX)
include(CTest)
get_filename_component(_quest_engine_default "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(WEBENGINE_SOURCE_DIR "${_quest_engine_default}" CACHE PATH "WebEngine source checkout")
option(WEBENGINE_BUILD_QUESTS "Build portable quest kernel" ON)
option(WEBENGINE_BUILD_QUEST_TESTS "Build quest kernel tests" ON)
add_subdirectory("${WEBENGINE_SOURCE_DIR}/modules" engine-modules)
add_subdirectory("${WEBENGINE_SOURCE_DIR}/quests" engine-quests)
if(NOT TARGET WebEngine::Quests)
  add_executable(quest_disabled ../modules/disabled.cpp)
  if(EMSCRIPTEN)
    set_target_properties(quest_disabled PROPERTIES SUFFIX ".js")
    target_link_options(quest_disabled PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
  endif()
  add_test(NAME quest_disabled COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:quest_disabled>)
  return()
endif()
webengine_add_module(example_quest_bindings ALIAS Example::QuestBindings
  PUBLIC_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}"
  SOURCES consumer_bindings.cpp PUBLIC_DEPENDENCIES WebEngine::Quests)
add_executable(quest_consumer consumer.cpp)
target_link_libraries(quest_consumer PRIVATE Example::QuestBindings)
if(EMSCRIPTEN)
  set_target_properties(quest_consumer PROPERTIES SUFFIX ".js")
  target_link_options(quest_consumer PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sSTACK_SIZE=1048576)
endif()
add_test(NAME quest_consumer COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:quest_consumer>)
