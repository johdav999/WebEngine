cmake_minimum_required(VERSION 3.22)
project(WebEngineSimulatorConsumer LANGUAGES CXX)
include(CTest)
set(WEBENGINE_SOURCE_DIR "" CACHE PATH "WebEngine checkout containing simulator/")
option(WEBENGINE_BUILD_SIMULATOR_EXAMPLE "Build standalone service and parity examples" OFF)
if(NOT EXISTS "${WEBENGINE_SOURCE_DIR}/simulator/CMakeLists.txt")
  message(FATAL_ERROR "Set WEBENGINE_SOURCE_DIR to a checkout with simulator/")
endif()
add_subdirectory("${WEBENGINE_SOURCE_DIR}/simulator" engine-simulator)
if(WEBENGINE_BUILD_SIMULATOR_EXAMPLE)
  if(NOT TARGET WebEngine::Simulator)
    message(FATAL_ERROR "Simulator example requires WEBENGINE_BUILD_SIMULATOR=ON")
  endif()
  add_executable(simulator_consumer headless.cpp)
  target_link_libraries(simulator_consumer PRIVATE WebEngine::Simulator)
  if(EMSCRIPTEN)
    set_target_properties(simulator_consumer PROPERTIES SUFFIX ".js")
    target_link_options(simulator_consumer PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
  endif()
  add_test(NAME simulator_consumer COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:simulator_consumer>)
endif()
foreach(forbidden engine_core engine_modules engine_rpg engine_quests Jolt engine_gui engine_ai engine_desktop)
  if(TARGET ${forbidden})
    message(FATAL_ERROR "Standalone simulator configured forbidden target: ${forbidden}")
  endif()
endforeach()
if(EXISTS "${CMAKE_BINARY_DIR}/_deps")
  message(FATAL_ERROR "Standalone simulator populated dependency downloads")
endif()

if(EMSCRIPTEN AND WEBENGINE_BUILD_SIMULATOR_TESTS)
  add_executable(simulator_browser_tests "${WEBENGINE_SOURCE_DIR}/tests/simulator_tests.cpp")
  target_link_libraries(simulator_browser_tests PRIVATE WebEngine::Simulator)
  set_target_properties(simulator_browser_tests PROPERTIES SUFFIX ".js")
  target_link_options(simulator_browser_tests PRIVATE -sENVIRONMENT=web -sEXIT_RUNTIME=1 -sALLOW_MEMORY_GROWTH=1)
  configure_file(browser.html browser.html COPYONLY)
endif()

if(WEBENGINE_BUILD_SIMULATOR_EXAMPLE)
  add_executable(simulator_parity parity.cpp)
  target_link_libraries(simulator_parity PRIVATE WebEngine::Simulator)
  if(EMSCRIPTEN)
    set_target_properties(simulator_parity PROPERTIES SUFFIX ".js")
    target_link_options(simulator_parity PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
  endif()
  add_test(NAME simulator_parity COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:simulator_parity>)
  add_test(NAME simulator_parity_partitioned COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:simulator_parity> partitioned)
endif()
