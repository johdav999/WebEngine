cmake_minimum_required(VERSION 3.22)
project(WebEngineModuleConsumer LANGUAGES CXX)
include(CTest)
set(WEBENGINE_SOURCE_DIR "" CACHE PATH "WebEngine checkout containing modules/")
if(NOT EXISTS "${WEBENGINE_SOURCE_DIR}/modules/CMakeLists.txt")
  message(FATAL_ERROR "Set WEBENGINE_SOURCE_DIR to a WebEngine checkout")
endif()
add_subdirectory("${WEBENGINE_SOURCE_DIR}/modules" engine-modules)
if(NOT TARGET WebEngine::Modules)
  add_executable(module_disabled disabled.cpp)
  if(EMSCRIPTEN)
    set_target_properties(module_disabled PROPERTIES SUFFIX ".js")
    target_link_options(module_disabled PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
  endif()
  add_test(NAME module_disabled COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:module_disabled>)
  return()
endif()
add_subdirectory(counter)
add_executable(module_consumer consumer.cpp)
target_link_libraries(module_consumer PRIVATE Example::Counter)
if(EMSCRIPTEN)
  set_target_properties(module_consumer PROPERTIES SUFFIX ".js")
  target_link_options(module_consumer PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
endif()
add_test(NAME module_consumer COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:module_consumer>)
