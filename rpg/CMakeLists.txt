# Standalone consumer: add modules/ then rpg/. No engine root configuration.
option(WEBENGINE_BUILD_RPG "Build portable RPG foundation" OFF)
option(WEBENGINE_BUILD_RPG_TESTS "Build RPG native/Wasm contract tests" OFF)
option(WEBENGINE_BUILD_RPG_DEMO "Build RPG rendered hosts (requires R9)" OFF)
option(WEBENGINE_BUILD_RPG_QUESTS "Build optional RPG quest bridge" OFF)
if(WEBENGINE_BUILD_RPG_DEMO)
  message(FATAL_ERROR "WEBENGINE_BUILD_RPG_DEMO is not implemented; rendered hosts are R9")
endif()
if(WEBENGINE_BUILD_RPG_QUESTS AND NOT WEBENGINE_BUILD_RPG)
  message(FATAL_ERROR "WEBENGINE_BUILD_RPG_QUESTS requires WEBENGINE_BUILD_RPG=ON")
endif()
if(NOT WEBENGINE_BUILD_RPG)
  if(WEBENGINE_BUILD_RPG_TESTS)
    message(FATAL_ERROR "WEBENGINE_BUILD_RPG_TESTS requires WEBENGINE_BUILD_RPG=ON")
  endif()
  return()
endif()
if(NOT TARGET WebEngine::Modules)
  message(FATAL_ERROR "WebEngine::RPG requires WebEngine::Modules; enable/add modules/ first")
endif()
webengine_add_module(engine_rpg ALIAS WebEngine::RPG
  PUBLIC_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../include"
  SOURCES ../src/rpg/snapshot.cpp ../src/rpg/snapshot_validation.cpp ../src/rpg/foundation.cpp ../src/rpg/characters.cpp ../src/rpg/runtime.cpp ../src/rpg/inventory.cpp ../src/rpg/random.cpp ../src/rpg/abilities.cpp ../src/rpg/encounters.cpp ../src/rpg/tactics.cpp
  PUBLIC_DEPENDENCIES WebEngine::Modules)
if(WEBENGINE_BUILD_RPG_QUESTS)
  if(NOT TARGET WebEngine::Quests)
    message(FATAL_ERROR "WebEngine::RPGQuests requires WebEngine::Quests; enable/add quests/ first")
  endif()
  webengine_add_module(engine_rpg_quests ALIAS WebEngine::RPGQuests
    PUBLIC_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../include"
    SOURCES ../src/rpg/quest_bridge.cpp
    PUBLIC_DEPENDENCIES WebEngine::RPG WebEngine::Quests)
endif()
