# Standalone consumer: add modules/ then rpg/. No engine root configuration.
option(WEBENGINE_BUILD_RPG "Build portable RPG foundation" OFF)
option(WEBENGINE_BUILD_RPG_TESTS "Build RPG native/Wasm contract tests" OFF)
option(WEBENGINE_BUILD_RPG_DEMO "Build RPG rendered hosts (requires R9)" OFF)
option(WEBENGINE_BUILD_RPG_QUESTS "Build RPG quest bridge (requires R7)" OFF)
if(WEBENGINE_BUILD_RPG_DEMO)
  message(FATAL_ERROR "WEBENGINE_BUILD_RPG_DEMO is not implemented; rendered hosts are R9")
endif()
if(WEBENGINE_BUILD_RPG_QUESTS)
  message(FATAL_ERROR "WEBENGINE_BUILD_RPG_QUESTS is not implemented; quest bridge is R7")
endif()
if(NOT WEBENGINE_BUILD_RPG)
  if(WEBENGINE_BUILD_RPG_TESTS)
    message(FATAL_ERROR "WEBENGINE_BUILD_RPG_TESTS requires WEBENGINE_BUILD_RPG=ON")
  endif()
  return()
endif()
if(NOT TARGET WebEngine::Modules)
  message(FATAL_ERROR "WebEngine::RPG requires WebEngine::Modules; enable/add modules/ first")
endif()
webengine_add_module(engine_rpg ALIAS WebEngine::RPG
  PUBLIC_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../include"
  SOURCES ../src/rpg/foundation.cpp ../src/rpg/characters.cpp ../src/rpg/runtime.cpp ../src/rpg/inventory.cpp ../src/rpg/random.cpp ../src/rpg/abilities.cpp
  PUBLIC_DEPENDENCIES WebEngine::Modules)
if(WEBENGINE_BUILD_RPG_TESTS)
  enable_testing()
  add_executable(rpg_foundation_tests ../tests/rpg_foundation_tests.cpp)
  target_link_libraries(rpg_foundation_tests PRIVATE WebEngine::RPG)
  if(EMSCRIPTEN)
    set_target_properties(rpg_foundation_tests PROPERTIES SUFFIX ".js")
    target_link_options(rpg_foundation_tests PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sALLOW_MEMORY_GROWTH=1)
  endif()
  add_test(NAME rpg_foundation COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:rpg_foundation_tests>)
  add_executable(rpg_character_tests ../tests/rpg_character_tests.cpp)
  target_link_libraries(rpg_character_tests PRIVATE WebEngine::RPG)
  if(EMSCRIPTEN)
    set_target_properties(rpg_character_tests PROPERTIES SUFFIX ".js")
    target_link_options(rpg_character_tests PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sALLOW_MEMORY_GROWTH=1)
  endif()
  add_test(NAME rpg_characters COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:rpg_character_tests>)
  add_executable(rpg_inventory_tests ../tests/rpg_inventory_tests.cpp)
  target_link_libraries(rpg_inventory_tests PRIVATE WebEngine::RPG)
  if(EMSCRIPTEN)
    set_target_properties(rpg_inventory_tests PROPERTIES SUFFIX ".js")
    target_link_options(rpg_inventory_tests PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sALLOW_MEMORY_GROWTH=1)
  endif()
  add_test(NAME rpg_inventory COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:rpg_inventory_tests>)
  add_executable(rpg_ability_tests ../tests/rpg_ability_tests.cpp)
  target_link_libraries(rpg_ability_tests PRIVATE WebEngine::RPG)
  if(EMSCRIPTEN)
    set_target_properties(rpg_ability_tests PROPERTIES SUFFIX ".js")
    target_link_options(rpg_ability_tests PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1 -sALLOW_MEMORY_GROWTH=1)
  endif()
  add_test(NAME rpg_abilities COMMAND ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:rpg_ability_tests>)
endif()
